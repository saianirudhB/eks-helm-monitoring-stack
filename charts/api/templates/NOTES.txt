  http{{ if .Values.ingress.tls.enabled }}s{{ end }}://{{ .Values.ingress.host }}{{ .Values.ingress.path }}

  kubectl get ingress -n {{ .Release.Namespace }} {{ include "api.fullname" . }}

  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "api.fullname" . }} 8080:{{ .Values.service.port }}
  curl http://localhost:8080/healthz
{{- end }}

4. Run the built-in Helm test (curls the health endpoint from inside the cluster):

  helm test {{ .Release.Name }} -n {{ .Release.Namespace }}
